"""隔离实证:旧名启动 → /reload-config 换新名/空串 → 重启前审批通知仍是旧命令。

用法: probe.py <startup_bot_name> <reload_bot_names...>
每个进程只模拟一次「启动」;要验重启后生效,用新名字再跑一次本脚本。
真实 misc_config 快照 + 真实 reload_runtime_config + 真实 handlers.commands matcher
注册期前缀,只在临时定制目录里写 YAML(不碰生产配置)。
"""

import os
import sys
import tempfile
from pathlib import Path

CUSTOM = Path(sys.argv[1])
STARTUP = sys.argv[2]
RELOADS = sys.argv[3:]
ROOT = Path(os.environ["REPO_ROOT"])
sys.path.insert(0, str(ROOT))
os.environ["AGENT_PLUGIN_CUSTOM_DIR"] = str(CUSTOM)
CUSTOM.mkdir(parents=True, exist_ok=True)
(CUSTOM / "misc_config.yaml").write_text(f"bot_name: {STARTUP!r}\n", encoding="utf-8")

from tests import _bootstrap  # noqa: E402,F401
from nonebot_plugin_hermes_adapter.core import approvals, misc_config  # noqa: E402
from nonebot_plugin_hermes_adapter.core.config_reload import (  # noqa: E402
    reload_runtime_config,
)
from nonebot_plugin_hermes_adapter.handlers import commands  # noqa: E402

entry = approvals.PendingApproval(
    approval_id=1,
    run_id="run-1",
    adapter="onebotv11",
    scope_desc="私聊 123",
    command="pwd",
    choices=["once", "deny"],
)


def report(tag: str) -> tuple[str, str]:
    notice = approvals.build_admin_notice(entry)
    prefix = approvals._approve_prefix()
    registered = commands._display_cmd("approve")
    assert commands.CMD_PREFIX == misc_config.startup_command_prefix()
    assert notice.count(prefix + "approve") == 4, notice
    print(
        f"[{tag}] snapshot_bot_name={misc_config.load_misc_config().bot_name!r} "
        f"CMD_PREFIX={commands.CMD_PREFIX!r} registered={registered!r} "
        f"notice_command={prefix + 'approve'!r} "
        f"notice==registered={prefix + 'approve' == registered}"
    )
    return prefix, notice


print(f"[startup] bot_name={STARTUP!r}")
registered_at_startup, _ = report("startup")
for new in RELOADS:
    (CUSTOM / "misc_config.yaml").write_text(f"bot_name: {new!r}\n", encoding="utf-8")
    result = reload_runtime_config()
    assert result.ok, result.message
    print(f"[reload -> {new!r}] {result.message}")
    prefix, notice = report(f"after reload {new!r}")
    assert misc_config.load_misc_config().bot_name == new, "快照必须真的换了"
    assert prefix == registered_at_startup, "重启前通知前缀必须仍是启动期注册的那条"
    assert prefix + "approve" == commands._display_cmd("approve")
    assert f"回复 {commands._display_cmd('approve')} " in notice
print(f"PASS startup={STARTUP!r} reloads={RELOADS} 重启前通知与注册命令一致")
